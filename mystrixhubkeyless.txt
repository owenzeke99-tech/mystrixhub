-- Tên file lưu trạng thái key trong thư mục của executor (nếu executor hỗ trợ)
local SAVE_FILE_NAME = "MystrixHub_KeyData.json"
local CORRECT_KEY = "MysticHubKeyHso - Hsiosownsosososjdjd"
local KEY_LINK = "https://link4m.org/4XXX3iqi"

-- Hàm kiểm tra xem đã lưu key và còn hạn 24h hay không
local function hasValidSavedKey()
	local success, result = pcall(function()
		if readfile and isfile and isfile(SAVE_FILE_NAME) then
			local content = readfile(SAVE_FILE_NAME)
			local data = HttpService:JSONDecode(content)
			-- Kiểm tra thời gian hiện tại so với lúc lưu (24 giờ = 86400 giây)
			if data and data.Key == CORRECT_KEY and (os.time() - data.Time) < 86400 then
				return true
			end
		end
	end)
	return success and result
end

-- Hàm lưu key kèm thời gian hiện tại
local function saveKeyData()
	pcall(function()
		if writefile then
			local data = {
				Key = CORRECT_KEY,
				Time = os.time()
			}
			writefile(SAVE_FILE_NAME, HttpService:JSONEncode(data))
		end
	end)
end

-- Hàm tải script chính của bạn
local function loadMainScript()
	local success, err = pcall(function()
		loadstring(game:HttpGet("https://raw.githubusercontent.com/ummarxfarooq/mystrix-hub/refs/heads/main/loader"))()
	end)
	if not success then
		warn("Error loading main script: " .. tostring(err))
	end
end

-- Nếu đã có key hợp lệ trong 24h, tự động load script luôn mà không hiện Menu

	loadMainScript()
	return
end

-- Hàm tạo hiệu ứng lắc mượt mà bằng TweenService
local function smoothShake()
	local originalPos = UDim2.new(0.5, -210, 0.5, -130)
	local shakeIntensity = 12
	
	for i = 1, 3 do
		local t1 = TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos + UDim2.new(0, shakeIntensity, 0, 0)})
		t1:Play()
		t1.Completed:Wait()
		
		local t2 = TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos - UDim2.new(0, shakeIntensity, 0, 0)})
		t2:Play()
		t2.Completed:Wait()
		
		shakeIntensity = shakeIntensity * 0.6
	end
	
	local resetTween = TweenService:Create(MainFrame, TweenInfo.new(0.05), {Position = originalPos})
	resetTween:Play()
end

		TextBox.Text = ""
		
		-- Chạy hiệu ứng lắc mượt
		smoothShake()
	end
end)
